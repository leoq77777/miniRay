cmake_minimum_required(VERSION 3.20)
project(MiniRay VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(Protobuf CONFIG QUIET)
if(NOT Protobuf_FOUND)
  find_package(Protobuf REQUIRED)
endif()
find_package(gRPC CONFIG REQUIRED)
set(PROTO_DIR ${CMAKE_CURRENT_SOURCE_DIR}/proto)
set(GENERATED_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GENERATED_DIR})
add_custom_command(
  OUTPUT ${GENERATED_DIR}/miniray.pb.cc ${GENERATED_DIR}/miniray.pb.h
         ${GENERATED_DIR}/miniray.grpc.pb.cc ${GENERATED_DIR}/miniray.grpc.pb.h
  COMMAND protobuf::protoc --proto_path=${PROTO_DIR} --cpp_out=${GENERATED_DIR}
          --grpc_out=${GENERATED_DIR} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin>
          ${PROTO_DIR}/miniray.proto
  DEPENDS ${PROTO_DIR}/miniray.proto protobuf::protoc gRPC::grpc_cpp_plugin VERBATIM)
add_library(miniray_proto ${GENERATED_DIR}/miniray.pb.cc ${GENERATED_DIR}/miniray.grpc.pb.cc)
target_include_directories(miniray_proto PUBLIC ${GENERATED_DIR})
target_link_libraries(miniray_proto PUBLIC protobuf::libprotobuf gRPC::grpc++)
add_library(miniray src/state.cpp src/coordinator.cpp src/worker.cpp src/client.cpp src/builtins.cpp)
target_include_directories(miniray PUBLIC include)
target_link_libraries(miniray PUBLIC miniray_proto Threads::Threads)
target_compile_options(miniray PRIVATE -Wall -Wextra -Wpedantic)
set(MINIRAY_SANITIZER "" CACHE STRING "Sanitizer: address or thread")
if(MINIRAY_SANITIZER)
  if(NOT MINIRAY_SANITIZER MATCHES "^(address|thread)$")
    message(FATAL_ERROR "MINIRAY_SANITIZER must be address or thread")
  endif()
  if(MINIRAY_SANITIZER STREQUAL "address")
    # Keep the message ABI compatible with prebuilt, non-ASan Protobuf libraries.
    target_compile_definitions(miniray_proto PUBLIC PROTOBUF_MESSAGE_GLOBALS_TEMPORARY_OPTOUT)
  endif()
  # Protobuf message layouts can depend on sanitizer macros; use identical flags everywhere.
  foreach(target miniray_proto miniray)
    target_compile_options(${target} PUBLIC -fsanitize=${MINIRAY_SANITIZER} -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=${MINIRAY_SANITIZER})
  endforeach()
endif()
add_executable(miniray-coordinator src/coordinator_main.cpp)
add_executable(miniray-worker src/worker_main.cpp)
add_executable(distributed-compute examples/distributed_compute.cpp)
add_executable(miniray-benchmark benchmarks/benchmark.cpp)
foreach(target miniray-coordinator miniray-worker distributed-compute miniray-benchmark)
  target_link_libraries(${target} PRIVATE miniray)
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(miniray-tests tests/runtime_test.cpp tests/rpc_test.cpp)
  target_link_libraries(miniray-tests PRIVATE miniray GTest::gtest_main)
  add_test(NAME runtime COMMAND miniray-tests)
  set_tests_properties(runtime PROPERTIES TIMEOUT 60)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME process-recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/process_recovery.py
    $<TARGET_FILE:miniray-coordinator> $<TARGET_FILE:miniray-worker> $<TARGET_FILE:distributed-compute>)
  set_tests_properties(process-recovery PROPERTIES TIMEOUT 45)
  add_test(NAME benchmark-accounting COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark_suite_test.py)
endif()
